const HB3_DEPT_FALLBACK=[
  {name:"פורמט רחב",color:"#62C7C2",has_rip:true,sort_order:1,bina_item_code:"8"},
  {name:"דיגיטלי צבעוני",color:"#3EA9A4",has_rip:true,sort_order:2,bina_item_code:"3"},
  {name:"דיגיטלי שחור לבן",color:"#6B7F92",has_rip:true,sort_order:3,bina_item_code:"4"},
  {name:"אופסט",color:"#0E3651",has_rip:true,sort_order:4,bina_item_code:"5"},
  {name:"ביגוד ומוצרי פרסום",color:"#EC008C",has_rip:true,sort_order:5,bina_item_code:"2"},
  {name:"מתקני תצוגה ומוצרים נלווים",color:"#00AEEF",has_rip:false,sort_order:6,bina_item_code:"7"},
  {name:"עבודות חוץ",color:"#D97706",has_rip:false,sort_order:7,bina_item_code:"6"},
];
const HB3_VAT=0.18;
function hb3EmptyItem(prev){
  return {description:"",quantity:1,unitPrice:0,lineTotal:0,discount:0,deptName:(prev&&prev.deptName)||"",binaItemCode:(prev&&prev.binaItemCode)||""};
}
function hb3LineTotal(it){
  const qty=Number(it.quantity)||0;
  const price=Number(it.unitPrice)||0;
  const disc=Number(it.discount)||0;
  return qty*price*(1-disc/100);
}
function hb3UnitFromTotal(it){
  const qty=Number(it.quantity)||0;
  const tot=Number(it.lineTotal)||0;
  const disc=Number(it.discount)||0;
  const denom=qty*(1-disc/100);
  if(!denom)return 0;
  return tot/denom;
}
function hb3FilledItems(items){
  return (items||[]).filter(it=>(it.description||"").trim());
}
function hb3ClientTerms(c){
  if(!c)return "";
  return c.default_payment_terms||c.payment_terms||"";
}
function hb3BuildPdfDoc({kind,binaDocId,title,client,contactName,items,agent,remark,paymentTerms,dueDate,priceMode}){
  return {
    docType:kind==="quote"?"quote":"order",
    binaDocId,
    title,
    client:{
      name:client?.name||"",
      address:client?.address||"",
      phone:client?.phone||"",
      email:client?.email||"",
      contactPerson:(contactName||"").trim()||client?.name||"",
    },
    items:(items||[]).map(it=>{
      const qty=Number(it.quantity)||1;
      const lineTotal=priceMode==="total"?Number(it.lineTotal)||hb3LineTotal(it):hb3LineTotal(it);
      const unitPrice=qty?lineTotal/qty:0;
      return {description:it.description,quantity:qty,unitPrice,discount:Number(it.discount)||0,lineTotal};
    }),
    agent:(agent||"").trim(),
    createdAt:new Date().toISOString(),
    remark:remark||"",
    paymentTerms:paymentTerms||"",
    dueDate:dueDate||"",
  };
}

function UnifiedOrderQuoteModal({open,onClose,onSuccess,clients,sb,initialMode="order",prefill=null,sourceQuoteId=null}){
  const [mode,setMode]=useState(initialMode==="quote"?"quote":"order");
  const [submitting,setSubmitting]=useState(false);
  const [error,setError]=useState(null);
  const [clientSearch,setClientSearch]=useState("");
  const [selectedClient,setSelectedClient]=useState(null);
  const [items,setItems]=useState([hb3EmptyItem()]);
  const [title,setTitle]=useState("");
  const [remark,setRemark]=useState("");
  const [agent,setAgent]=useState("");
  const [contactName,setContactName]=useState("");
  const [dueDate,setDueDate]=useState("");
  const [workFile,setWorkFile]=useState(null);
  const [workerNote,setWorkerNote]=useState("");
  const [activeQuoteId,setActiveQuoteId]=useState(sourceQuoteId||null);
  const [priceMode,setPriceMode]=useState("unit");
  const [paymentTerms,setPaymentTerms]=useState("");
  const [depts,setDepts]=useState(HB3_DEPT_FALLBACK);
  const [success,setSuccess]=useState(null);
  const [dupNote,setDupNote]=useState(null);
  const [duplicatedFrom,setDuplicatedFrom]=useState(null);

  const loadDepts=async()=>{
    try{
      const{data,error:e}=await supabaseClient.from("department_settings").select("*").order("sort_order");
      if(!e&&data?.length)setDepts(data);
    }catch(err){console.warn(err);}
  };

  const applyPrefill=async(p)=>{
    if(!p)return;
    if(p.mode)setMode(p.mode==="quote"?"quote":"order");
    if(p.title!=null)setTitle(p.title||"");
    if(p.remark!=null)setRemark(p.remark||"");
    if(p.agent!=null)setAgent(p.agent||"");
    if(p.contactName!=null)setContactName(p.contactName||"");
    if(p.dueDate!=null)setDueDate(p.dueDate||"");
    if(p.workerNote!=null)setWorkerNote(p.workerNote||"");
    if(p.priceMode)setPriceMode(p.priceMode==="total"?"total":"unit");
    if(p.paymentTerms!=null)setPaymentTerms(p.paymentTerms||"");
    if(p.duplicatedFromOrderId)setDuplicatedFrom(String(p.duplicatedFromOrderId));
    if(p.sourceCreatedAt){
      const age=(Date.now()-new Date(p.sourceCreatedAt).getTime())/864e5;
      if(age>60){
        const d=new Date(p.sourceCreatedAt).toLocaleDateString("he-IL");
        setDupNote("ההזמנה המקורית מ־"+d+" — כדאי לוודא שהמחירים עדיין מעודכנים");
      }
    }
    if(Array.isArray(p.items)&&p.items.length){
      const mapped=p.items.map(it=>{
        const qty=it.quantity!=null?it.quantity:1;
        const unit=it.unitPrice!=null?it.unitPrice:(it.unit_price!=null?it.unit_price:0);
        const disc=it.discount!=null?it.discount:(it.discount_pct!=null?it.discount_pct:0);
        const row={description:it.description||"",quantity:qty,unitPrice:unit,discount:disc,deptName:it.deptName||it.department||"",binaItemCode:it.binaItemCode||it.bina_item_code||""};
        row.lineTotal=it.lineTotal!=null?it.lineTotal:hb3LineTotal(row);
        return row;
      });
      mapped.push(hb3EmptyItem(mapped[mapped.length-1]));
      setItems(mapped);
    }
    if(p.client){
      setSelectedClient(p.client);
      if(p.paymentTerms==null)setPaymentTerms(hb3ClientTerms(p.client));
    }else if(p.bina_cust_id!=null){
      const list=Array.isArray(clients)?clients:[];
      const found=list.find(c=>String(c.bina_customer_id)===String(p.bina_cust_id));
      if(found){
        setSelectedClient(found);
        if(p.paymentTerms==null)setPaymentTerms(hb3ClientTerms(found));
      }
    }
    if(p.sourceQuoteId)setActiveQuoteId(p.sourceQuoteId);
  };

  useEffect(()=>{
    if(!open)return;
    setMode(initialMode==="quote"?"quote":"order");
    setClientSearch("");
    setSelectedClient(null);
    setItems([hb3EmptyItem()]);
    setTitle("");
    setRemark("");
    setAgent("");
    setContactName("");
    setDueDate("");
    setWorkFile(null);
    setWorkerNote("");
    setActiveQuoteId(sourceQuoteId||null);
    setError(null);
    setSubmitting(false);
    setSuccess(null);
    setDupNote(null);
    setDuplicatedFrom(null);
    setPriceMode("unit");
    setPaymentTerms("");
    loadDepts();
    if(prefill)applyPrefill(prefill);
  },[open,initialMode,sourceQuoteId]);

  if(!open)return null;

  const isOrder=mode==="order";
  const filled=hb3FilledItems(items);
  const itemsTotal=filled.reduce((sum,it)=>sum+(priceMode==="total"?Number(it.lineTotal)||hb3LineTotal(it):hb3LineTotal(it)),0);
  const vatAmt=itemsTotal*HB3_VAT;
  const grand=itemsTotal+vatAmt;
  const filteredClients=(clients||[]).filter(c=>{
    if(!clientSearch.trim())return true;
    const q=clientSearch.toLowerCase().trim();
    return (c.name||"").toLowerCase().includes(q)||(c.phone||"").includes(q)||String(c.bina_customer_id||"").includes(q);
  }).slice(0,50);

  const ensureTrailing=rows=>{
    const last=rows[rows.length-1];
    if(last&&(last.description||"").trim())rows.push(hb3EmptyItem(last));
    return rows;
  };
  const updateItem=(idx,field,value)=>{
    setItems(prev=>{
      const ni=prev.map((it,i)=>i===idx?{...it,[field]:value}:it);
      const it=ni[idx];
      if(field==="deptName"){
        const d=depts.find(x=>x.name===value);
        it.binaItemCode=d?.bina_item_code||"";
      }
      if(priceMode==="unit"&&(field==="quantity"||field==="unitPrice"||field==="discount"))it.lineTotal=hb3LineTotal(it);
      if(priceMode==="total"&&(field==="quantity"||field==="lineTotal"||field==="discount"))it.unitPrice=hb3UnitFromTotal(it);
      return ensureTrailing(ni);
    });
  };
  const removeItem=idx=>{
    setItems(prev=>{
      const ni=prev.filter((_,i)=>i!==idx);
      return ni.length?ensureTrailing(ni):[hb3EmptyItem()];
    });
  };
  const pickClient=c=>{
    if(c.bina_customer_id){
      setSelectedClient(c);
      setPaymentTerms(hb3ClientTerms(c));
    }else{
      const ok=window.confirm("הלקוח לא מקושר לבינה. לפתוח יצירת לקוח?");
      if(ok){onClose();if(window.openBinaClient)window.openBinaClient();}
    }
  };
  const switchPriceMode=m=>{
    setPriceMode(m);
    setItems(prev=>prev.map(it=>{
      const copy={...it};
      if(m==="total")copy.lineTotal=hb3LineTotal(copy);
      else copy.unitPrice=hb3UnitFromTotal(copy)||copy.unitPrice;
      return copy;
    }));
  };

  const validate=()=>{
    if(!selectedClient)return "נא לבחור לקוח";
    if(!selectedClient.bina_customer_id)return "הלקוח לא מקושר לבינה";
    if(!filled.length)return "נא למלא לפחות פריט אחד";
    if(isOrder&&!dueDate)return "נא למלא דדליין להזמנה";
    if(isOrder&&!workFile&&!prefill?.skipFileRequired)return "נא לצרף קובץ עבודה להזמנה";
    return null;
  };

  const attachWorkFileAfterOrder=async(binaOrderId,taskIdHint)=>{
    if(!workFile)return;
    const year=new Date().getFullYear();
    const uploaded=await uploadFileToStorage(workFile,"work-files/"+year);
    const meta={
      file_path:uploaded.path,
      file_name:workFile.name,
      file_type:workFile.type||null,
      file_size_kb:Math.round((workFile.size||0)/1024),
      worker_note:workerNote.trim()||null,
      due_date:dueDate||null,
    };
    let taskId=taskIdHint||null;
    if(!taskId){
      const{data:found}=await supabaseClient.from("tasks").select("id").eq("bina_order_id",String(binaOrderId)).is("archived_at",null).limit(1);
      taskId=found?.[0]?.id||null;
    }
    if(taskId)await applyWorkFileToTask(taskId,meta);
    else{
      const{error:pErr}=await supabaseClient.from("pending_work_files").insert({
        created_by_name:null,
        bina_order_id:String(binaOrderId),
        file_path:meta.file_path,
        file_name:meta.file_name,
        file_type:meta.file_type,
        file_size_kb:meta.file_size_kb,
        worker_note:meta.worker_note,
        due_date:meta.due_date,
        status:"waiting",
      });
      if(pErr)throw pErr;
    }
  };

  const saveClientTerms=async()=>{
    if(!selectedClient?.id)return;
    const val=(paymentTerms||"").trim()||null;
    try{
      await supabaseClient.from("clients").update({
        default_payment_terms:val,
        payment_terms:val,
        payment_terms_updated_at:new Date().toISOString(),
      }).eq("id",selectedClient.id);
    }catch(e){console.warn("payment terms save",e);}
  };

  const binaItems=()=>filled.map((it,idx)=>{
    const qty=Number(it.quantity)||1;
    const lineTotal=priceMode==="total"?Number(it.lineTotal)||hb3LineTotal(it):hb3LineTotal(it);
    const unitPrice=qty?lineTotal/qty:0;
    return {
      itemId:String(it.binaItemCode||("LINE-"+(idx+1))),
      description:it.description||("פריט "+(idx+1)),
      quantity:qty,
      unitPrice,
      discount:Number(it.discount)||0,
    };
  });

  const localLineRows=kindId=>(idKey,idVal)=>filled.map((it,idx)=>{
    const qty=Number(it.quantity)||1;
    const lineTotal=priceMode==="total"?Number(it.lineTotal)||hb3LineTotal(it):hb3LineTotal(it);
    const unitPrice=qty?lineTotal/qty:0;
    const row={
      [idKey]:idVal,
      line_number:idx+1,
      description:it.description||("פריט "+(idx+1)),
      quantity:qty,
      unit_price:unitPrice,
      discount_pct:Number(it.discount)||0,
      price_entry_mode:priceMode,
      bina_item_code:it.binaItemCode||null,
      line_total:lineTotal,
    };
    if(kindId==="quote")row.item_name=String(it.binaItemCode||("LINE-"+(idx+1)));
    return row;
  });

  const submitOrder=async()=>{
    const payload={
      client:{
        binaCustomerId:String(selectedClient.bina_customer_id),
        name:selectedClient.name||"ללא שם",
        city:selectedClient.address||"ללא",
        address:selectedClient.address||"ללא",
        contactPerson:(contactName||"").trim()||selectedClient.name||"ללא",
        phone:selectedClient.phone||"",
        email:selectedClient.email||"",
      },
      items:binaItems(),
      title:title||("הזמנה ל"+(selectedClient.name||"לקוח")),
      remark:remark||"",
    };
    const{data,error:invE}=await supabaseClient.functions.invoke("bina-create-order",{body:payload});
    if(invE)throw new Error(invE.message||String(invE));
    let result=data;
    if(typeof data==="string"){try{result=JSON.parse(data);}catch{result=null;}}
    if(!result||result.success!==true)throw new Error(result?.error||"שגיאה ביצירת הזמנה בבינה");

    if(result.taskId){
      const taskPatch={sales_agent:(agent||"").trim()||null};
      const ag=(agent||"").trim();
      if(ag)taskPatch.assignee=ag;
      if(dueDate)taskPatch.due_date=dueDate;
      if(duplicatedFrom)taskPatch.duplicated_from_order_id=duplicatedFrom;
      try{await sb("tasks?id=eq."+result.taskId,{method:"PATCH",body:JSON.stringify(taskPatch)});}catch(e){console.warn(e);}
    }
    try{
      const addrStr=selectedClient.address||"";
      const cityPart=(addrStr.split(",")[0]||"").trim();
      const orderRow={
        bina_doc_id:Number(result.binaOrderId),
        title:title||("הזמנה ל"+(selectedClient.name||"לקוח")),
        bina_cust_id:parseInt(String(selectedClient.bina_customer_id),10),
        bina_cust_name:selectedClient.name||"",
        bina_cust_address:addrStr||null,
        bina_cust_city:cityPart||null,
        bina_cust_phone:selectedClient.phone||null,
        bina_cust_email:selectedClient.email||null,
        contact_person:(contactName||"").trim()||null,
        sales_agent:(agent||"").trim()||null,
        remark:remark||"",
        status:"נשלחה",
        task_id:result.taskId||null,
      };
      const ordRes=await sb("orders",{method:"POST",headers:{Prefer:"return=representation"},body:JSON.stringify(orderRow)});
      const ord=Array.isArray(ordRes)?ordRes[0]:ordRes;
      if(ord?.id){
        await sb("order_items",{method:"POST",body:JSON.stringify(localLineRows("order")("order_id",ord.id))});
      }
    }catch(e){console.warn("[UnifiedOrderQuoteModal] orders DB save failed:",e);}

    try{await attachWorkFileAfterOrder(result.binaOrderId,result.taskId||null);}
    catch(fileErr){
      console.warn("work file attach failed",fileErr);
      window.alert("ההזמנה נוצרה, אך חיבור הקובץ נכשל: "+(fileErr.message||fileErr)+" — אפשר לחבר בטאב קבצי עבודה");
    }
    if(activeQuoteId){
      try{
        await supabaseClient.from("quotes").update({
          quote_status:"approved",
          converted_to_order_id:String(result.binaOrderId),
          converted_at:new Date().toISOString(),
          status_changed_at:new Date().toISOString(),
        }).eq("id",activeQuoteId);
      }catch(e){console.warn(e);}
    }
    await saveClientTerms();
    const pdfDoc=hb3BuildPdfDoc({kind:"order",binaDocId:result.binaOrderId,title:payload.title,client:selectedClient,contactName,items:filled,agent,remark,paymentTerms,dueDate,priceMode});
    if(onSuccess)onSuccess(result);
    setSuccess({kind:"order",id:result.binaOrderId,pdfDoc});
  };

  const submitQuote=async()=>{
    const quoteTitle=title||("הצעה ל"+(selectedClient.name||"לקוח"));
    const binaCustNum=parseInt(String(selectedClient.bina_customer_id),10);
    if(!binaCustNum||Number.isNaN(binaCustNum))throw new Error("קוד לקוח בינה לא תקין");
    const quoteData={
      title:quoteTitle,
      bina_cust_id:binaCustNum,
      bina_cust_name:selectedClient.name||"ללא שם",
      bina_cust_city:(selectedClient.address||"").split(",")[0]?.trim()||selectedClient.address||"",
      bina_cust_address:selectedClient.address||"",
      bina_cust_phone:selectedClient.phone||"",
      contact_person:(contactName||"").trim(),
      sales_agent:agent||null,
      status:"טיוטה",
      quote_status:"sent",
      sent_at:new Date().toISOString(),
    };
    let quoteId=null;
    try{
      const quoteResponse=await sb("quotes",{method:"POST",headers:{Prefer:"return=representation"},body:JSON.stringify(quoteData)});
      const quoteResult=Array.isArray(quoteResponse)?quoteResponse[0]:quoteResponse;
      if(!quoteResult?.id)throw new Error("שגיאה ביצירת הצעה ב-DB");
      quoteId=quoteResult.id;
      await sb("quote_items",{method:"POST",body:JSON.stringify(localLineRows("quote")("quote_id",quoteId))});
      const{data:fnData,error:invE}=await supabaseClient.functions.invoke("bina-create-quote",{body:{quote_id:quoteId}});
      if(invE)throw new Error(invE.message||String(invE));
      let result=fnData;
      if(typeof fnData==="string"){try{result=JSON.parse(fnData);}catch{result=null;}}
      if(!result||result.success!==true)throw new Error(result?.error||result?.binaMessage||"שגיאה בשליחה לבינה");
      try{
        await supabaseClient.from("quotes").update({quote_status:"sent",sent_at:new Date().toISOString()}).eq("id",quoteId);
      }catch(e){console.warn(e);}
      await saveClientTerms();
      const pdfDoc=hb3BuildPdfDoc({kind:"quote",binaDocId:result.binaDocId,title:quoteTitle,client:selectedClient,contactName,items:filled,agent,remark,paymentTerms,dueDate:"",priceMode});
      if(onSuccess)onSuccess(result);
      setSuccess({kind:"quote",id:result.binaDocId,pdfDoc});
    }catch(err){
      if(quoteId){
        try{await sb("quote_items?quote_id=eq."+quoteId,{method:"DELETE"});}catch(delE){console.warn("rollback quote_items failed",delE);}
        try{await sb("quotes?id=eq."+quoteId,{method:"DELETE"});}catch(delE){console.warn("rollback quote failed",delE);}
      }
      throw err;
    }
  };

  const handleSubmit=async()=>{
    const v=validate();
    if(v){setError(v);return;}
    const missing=filled.map((it,i)=>!(it.binaItemCode||"").trim()?i+1:null).filter(Boolean);
    if(missing.length){
      const ok=window.confirm("פריט "+missing.join(", ")+" בלי מחלקה — לא יופיע אצל אף עובד. להמשיך?");
      if(!ok)return;
    }
    setSubmitting(true);
    setError(null);
    try{
      if(isOrder)await submitOrder();
      else await submitQuote();
    }catch(err){
      setError(err.message||String(err));
    }finally{
      setSubmitting(false);
    }
  };

  const overlayStyle={position:"fixed",inset:0,background:"rgba(14,54,81,0.45)",zIndex:9999,display:"flex",alignItems:"flex-end",justifyContent:"center"};
  const modalStyle={width:"100%",maxWidth:860,maxHeight:"96vh",background:"#F7FBFA",borderRadius:"16px 16px 0 0",display:"flex",flexDirection:"column",boxShadow:"0 -8px 32px rgba(14,54,81,0.18)",overflow:"hidden"};

  if(success){
    const lab=success.kind==="order"?"הזמנה":"הצעה";
    return<div style={overlayStyle} onClick={()=>!submitting&&onClose()}>
      <div style={modalStyle} onClick={e=>e.stopPropagation()}>
        <div style={{padding:16,display:"flex",justifyContent:"space-between",alignItems:"center"}}>
          <strong style={{color:"#0E3651"}}>נוצר בבינה</strong>
          <button type="button" onClick={onClose} style={{background:"none",border:"none",fontSize:22,cursor:"pointer",fontFamily:"inherit"}}>✕</button>
        </div>
        <div style={{padding:"8px 16px 24px"}}>
          <div className="hb3-ok">
            <div style={{display:"flex",justifyContent:"space-between",gap:12,flexWrap:"wrap",alignItems:"center"}}>
              <div>
                <div style={{fontWeight:800,color:"#0E3651",fontSize:16}}>✅ {lab} {success.id} נוצרה בבינה</div>
                {success.kind==="order"&&<div style={{fontSize:13,color:"#2A807C",marginTop:4}}>הקובץ יחובר למשימה אוטומטית כשהסנכרון יסתיים</div>}
              </div>
              <button type="button" onClick={()=>window.openDocumentPDF&&window.openDocumentPDF(success.pdfDoc)} style={{padding:"10px 14px",borderRadius:10,border:"none",background:"#62C7C2",color:"#fff",fontWeight:800,cursor:"pointer",fontFamily:"inherit",minHeight:44}}>📄 הדפס PDF</button>
            </div>
          </div>
        </div>
      </div>
    </div>;
  }

  return<div style={overlayStyle} onClick={()=>!submitting&&onClose()}>
    <div style={modalStyle} onClick={e=>e.stopPropagation()}>
      <div style={{padding:"12px 16px",display:"flex",alignItems:"center",justifyContent:"space-between",gap:8,background:"#fff",borderBottom:"1px solid #E2EEED"}}>
        <div className="hb3-toggle" style={{maxWidth:280,flex:1}}>
          <button type="button" className={!isOrder?"on":""} onClick={()=>!submitting&&setMode("quote")}>הצעת מחיר</button>
          <button type="button" className={isOrder?"on":""} onClick={()=>!submitting&&setMode("order")}>הזמנה</button>
        </div>
        <button type="button" onClick={onClose} disabled={submitting} style={{background:"none",border:"none",fontSize:22,cursor:"pointer",color:"#0E3651",fontFamily:"inherit"}}>✕</button>
      </div>
      <div style={{flex:1,overflowY:"auto",padding:14,direction:"rtl"}}>
        {isOrder&&<div style={{background:"#FFFBEB",border:"1px solid #FDE68A",borderRight:"4px solid #D97706",borderRadius:10,padding:"10px 12px",marginBottom:12,fontSize:13,color:"#92400E"}}>
          <strong>שווה לוודא את תנאי התשלום</strong>
          <div style={{marginTop:4}}>הם נמשכים מכרטיס הלקוח וקובעים את מועד הגבייה. שינוי כאן מתעדכן בכרטיס הלקוח ויחול על כל ההזמנות הבאות.</div>
        </div>}
        {dupNote&&<div style={{background:"#EDF8F7",border:"1px solid #BFE6E3",borderRadius:10,padding:"10px 12px",marginBottom:12,fontSize:13,color:"#0E3651"}}>ℹ️ {dupNote}</div>}

        <div className="hb3-grid" style={{marginBottom:12}}>
          <div className="hb3-card">
            <div className="hb3-sec">לקוח</div>
            {selectedClient?<div>
              <div style={{fontWeight:800,color:"#0E3651"}}>{selectedClient.name}</div>
              <div style={{fontSize:12,color:"#2A807C",marginTop:4}}>{selectedClient.phone||"—"} · {selectedClient.bina_customer_id}</div>
              <button type="button" onClick={()=>setSelectedClient(null)} style={{marginTop:8,padding:"6px 10px",borderRadius:8,border:"1px solid #D5E3E1",background:"#fff",cursor:"pointer",fontFamily:"inherit"}}>החלף</button>
            </div>:<>
              <input className="hb3-inp" type="text" placeholder="חיפוש שם / טלפון / קוד בינה" value={clientSearch} onChange={e=>setClientSearch(e.target.value)}/>
              <div style={{marginTop:8,maxHeight:140,overflowY:"auto"}}>
                {filteredClients.map(c=><div key={c.id} onClick={()=>pickClient(c)} style={{padding:"8px 10px",marginBottom:4,borderRadius:8,cursor:"pointer",background:c.bina_customer_id?"#EDF8F7":"#FFFBEB"}}>
                  <div style={{fontWeight:700,fontSize:13}}>{c.name||"ללא שם"}</div>
                  <div style={{fontSize:11,color:"#6B7F92"}}>{c.phone||"—"}</div>
                </div>)}
              </div>
            </>}
          </div>
          <div className="hb3-card">
            <div className="hb3-sec">פרטי מסמך</div>
            <label className="hb3-lab">כותרת</label>
            <input className="hb3-inp" style={{marginBottom:8}} value={title} onChange={e=>setTitle(e.target.value)} placeholder={isOrder?"כותרת הזמנה":"כותרת הצעה"}/>
            <label className="hb3-lab">סוכן</label>
            <select className="hb3-inp" style={{marginBottom:8}} value={agent} onChange={e=>setAgent(e.target.value)}>
              <option value="">בחר סוכן...</option>
              <option value="כפיר">כפיר</option>
              <option value="נטלי">נטלי</option>
              <option value="ברק">ברק</option>
            </select>
            <label className="hb3-lab">לידי</label>
            <input className="hb3-inp" style={{marginBottom:8}} value={contactName} onChange={e=>setContactName(e.target.value)}/>
            <label className="hb3-lab">תנאי תשלום</label>
            <input className="hb3-inp" value={paymentTerms} onChange={e=>setPaymentTerms(e.target.value)} placeholder="שוטף + 30"/>
            {selectedClient&&<div style={{fontSize:11,color:"#6B7F92",marginTop:4}}>מכרטיס הלקוח · ניתן לשנות</div>}
            {isOrder&&<>
              <label className="hb3-lab" style={{marginTop:8}}>הערות</label>
              <textarea className="hb3-inp" rows={2} value={remark} onChange={e=>setRemark(e.target.value)} style={{resize:"vertical"}}/>
            </>}
          </div>
        </div>

        <div className="hb3-card" style={{marginBottom:12}}>
          <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",gap:8,flexWrap:"wrap",marginBottom:8}}>
            <div className="hb3-sec" style={{margin:0}}>פריטים</div>
            <div className="hb3-toggle" style={{width:220}}>
              <button type="button" className={priceMode==="unit"?"on":""} onClick={()=>switchPriceMode("unit")}>מחיר יחידה</button>
              <button type="button" className={priceMode==="total"?"on":""} onClick={()=>switchPriceMode("total")}>סה״כ</button>
            </div>
          </div>
          {items.map((it,idx)=>{
            const isEmpty=!(it.description||"").trim()&&idx===items.length-1;
            const computedUnit=hb3UnitFromTotal(it);
            const computedTot=hb3LineTotal(it);
            return<div key={idx} className={"hb3-item"+(isEmpty?" empty":"")}>
              <div style={{display:"flex",gap:8,alignItems:"center"}}>
                <input className="hb3-inp" value={it.description} onChange={e=>updateItem(idx,"description",e.target.value)} placeholder="תיאור הפריט"/>
                {items.length>1&&<button type="button" onClick={()=>removeItem(idx)} style={{border:"none",background:"#FEE2E2",color:"#DC2626",borderRadius:8,width:36,height:36,cursor:"pointer",fontFamily:"inherit"}}>×</button>}
              </div>
              <div className="hb3-item-grid">
                <div>
                  <label className="hb3-lab">מחלקה</label>
                  <select className="hb3-inp" value={it.deptName||""} onChange={e=>updateItem(idx,"deptName",e.target.value)}>
                    <option value="">בחר...</option>
                    {depts.map(d=><option key={d.name} value={d.name}>{d.name}</option>)}
                  </select>
                  {it.deptName&&<span style={{display:"inline-block",width:8,height:8,borderRadius:"50%",background:(depts.find(d=>d.name===it.deptName)||{}).color||"#62C7C2",marginTop:6}}/>}
                </div>
                <div>
                  <label className="hb3-lab">כמות</label>
                  <input className="hb3-inp" type="number" value={it.quantity} onChange={e=>updateItem(idx,"quantity",e.target.value)}/>
                </div>
                <div>
                  <label className="hb3-lab">מחיר יחידה</label>
                  {priceMode==="unit"?
                    <input className="hb3-inp" type="number" step="0.001" value={it.unitPrice} onChange={e=>updateItem(idx,"unitPrice",e.target.value)}/>:
                    <div>
                      <input className="hb3-inp" readOnly value={computedUnit.toFixed(3)} style={{background:"#F1F5F4",color:"#6B7F92"}}/>
                      <div style={{fontSize:10,color:"#6B7F92",marginTop:2}}>מחושב</div>
                    </div>}
                </div>
                <div>
                  <label className="hb3-lab">סה״כ</label>
                  {priceMode==="total"?
                    <input className="hb3-inp" type="number" step="0.01" value={it.lineTotal} onChange={e=>updateItem(idx,"lineTotal",e.target.value)}/>:
                    <input className="hb3-inp" readOnly value={computedTot.toFixed(2)} style={{background:"#F1F5F4",color:"#6B7F92"}}/>}
                </div>
              </div>
            </div>;
          })}
        </div>

        {isOrder&&<div className="hb3-card" style={{marginBottom:12}}>
          <div className="hb3-sec">קובץ ודדליין</div>
          <div className="hb3-grid">
            <div>
              <label className="hb3-lab">דדליין *</label>
              <input className="hb3-inp" type="date" value={dueDate} onChange={e=>setDueDate(e.target.value)}/>
            </div>
            <div>
              <label className="hb3-lab">קובץ עבודה {prefill?.skipFileRequired?"":"*"}</label>
              <input type="file" onChange={e=>setWorkFile(e.target.files?.[0]||null)}/>
              {workFile&&<div style={{fontSize:12,color:"#2A807C",marginTop:4}}>{workFile.name}</div>}
            </div>
          </div>
          <label className="hb3-lab" style={{marginTop:8}}>הערה לעובד</label>
          <textarea className="hb3-inp" rows={2} value={workerNote} onChange={e=>setWorkerNote(e.target.value)} style={{resize:"vertical"}}/>
        </div>}

        {error&&<div style={{padding:12,background:"#FEF2F2",color:"#DC2626",borderRadius:10,fontSize:14,marginBottom:8}}>
          <div>❌ {error}</div>
          <div style={{fontSize:12,marginTop:6,color:"#991B1B"}}>הטופס נשמר — אפשר לנסות שוב אחרי תיקון.</div>
        </div>}
      </div>
      <div className="hb3-sticky">
        <div style={{display:"flex",gap:16,flexWrap:"wrap",fontSize:13}}>
          <span>לפני מע״מ <strong>{itemsTotal.toFixed(2)}</strong></span>
          <span>מע״מ <strong>{vatAmt.toFixed(2)}</strong></span>
          <span style={{fontSize:16,fontWeight:800}}>סה״כ ₪{grand.toFixed(2)}</span>
        </div>
        <button type="button" onClick={handleSubmit} disabled={submitting} style={{padding:"12px 22px",border:"none",borderRadius:10,background:submitting?"#9CA3AF":"#62C7C2",color:"#fff",fontWeight:800,cursor:"pointer",fontFamily:"inherit",minHeight:44,minWidth:120}}>{submitting?"שולח...":(isOrder?"צור הזמנה":"צור הצעה")}</button>
      </div>
    </div>
  </div>;
}

function QuotesLifecycleTab({clients,sb,onConvertToOrder,onDuplicateOrder}){
  const [quotes,setQuotes]=useState([]);
  const [orders,setOrders]=useState([]);
  const [loading,setLoading]=useState(true);
  const [viewQ,setViewQ]=useState(null);
  const [busyId,setBusyId]=useState(null);
  const [histQ,setHistQ]=useState("");

  const load=async()=>{
    setLoading(true);
    try{
      const[{data,error},{data:ords,error:oErr}]=await Promise.all([
        supabaseClient.from("quotes").select("id,title,bina_doc_id,bina_cust_id,bina_cust_name,contact_person,sales_agent,quote_status,sent_at,created_at,follow_up_sent_at,converted_to_order_id,converted_at,status_changed_at,total_amount,status").order("sent_at",{ascending:true,nullsFirst:false}),
        supabaseClient.from("orders").select("id,title,bina_doc_id,bina_cust_id,bina_cust_name,contact_person,sales_agent,created_at,remark,order_items(id,description,quantity,unit_price,discount_pct,bina_item_code,price_entry_mode,line_total,line_number)").order("created_at",{ascending:false}).limit(250),
      ]);
      if(error)throw error;
      if(oErr)console.warn(oErr);
      setQuotes(data||[]);
      setOrders(ords||[]);
    }catch(e){console.warn(e);setQuotes([]);}
    setLoading(false);
  };
  useEffect(()=>{load();},[]);

  const ageDays=q=>{
    const base=q.sent_at||q.created_at;
    if(!base)return 0;
    return Math.floor((Date.now()-new Date(base).getTime())/864e5);
  };
  const ageColor=d=>d>=15?"#dc2626":d>=8?"#ea580c":"#64748b";
  const openQuotes=quotes.filter(q=>(q.quote_status||"sent")==="sent");
  const now=new Date();
  const monthStart=new Date(now.getFullYear(),now.getMonth(),1).toISOString();
  const approvedMonth=quotes.filter(q=>q.quote_status==="approved"&&q.converted_at&&q.converted_at>=monthStart).length;
  const approvedMonthAlt=quotes.filter(q=>q.quote_status==="approved"&&(q.status_changed_at||q.converted_at||q.created_at)>=monthStart).length;
  const approvedCount=Math.max(approvedMonth,approvedMonthAlt);
  const rejectedCount=quotes.filter(q=>q.quote_status==="rejected").length;
  const closedLike=approvedCount+rejectedCount;
  const convRate=closedLike?Math.round((approvedCount/closedLike)*100):0;

  const setStatus=async(q,status)=>{
    setBusyId(q.id);
    try{
      const{error}=await supabaseClient.from("quotes").update({quote_status:status,status_changed_at:new Date().toISOString()}).eq("id",q.id);
      if(error)throw error;
      await load();
    }catch(e){window.alert(e.message||String(e));}
    finally{setBusyId(null);}
  };

  const sendFollowUp=async(q)=>{
    setBusyId(q.id);
    try{
      await sendEmail({
        to:["kfir.dfus@gmail.com","natalie.zem@gmail.com"],
        subject:"תזכורת הצעה #"+((q.bina_doc_id||q.id)+"").slice(0,12)+" — "+(q.bina_cust_name||""),
        html:"<!DOCTYPE html><html dir=\"rtl\" lang=\"he\"><body style=\"font-family:Arial,sans-serif;direction:rtl;padding:16px\">"+
          "<h2>תזכורת להצעת מחיר</h2>"+
          "<p><strong>לקוח:</strong> "+escHtml(q.bina_cust_name||"—")+"</p>"+
          "<p><strong>כותרת:</strong> "+escHtml(q.title||"—")+"</p>"+
          "<p><strong>מספר בבינה:</strong> "+escHtml(q.bina_doc_id||"—")+"</p>"+
          "<p><strong>ימים מאז שליחה:</strong> "+ageDays(q)+"</p>"+
          "</body></html>",
        from:"PrintOS <orders@natalie-print.com>",
      });
      await supabaseClient.from("quotes").update({follow_up_sent_at:new Date().toISOString()}).eq("id",q.id);
      await load();
      window.alert("תזכורת נשלחה");
    }catch(e){window.alert(e.message||String(e));}
    finally{setBusyId(null);}
  };

  const mapQuoteItems=qItems=>(qItems||[]).map(it=>{
    const d=HB3_DEPT_FALLBACK.find(x=>x.bina_item_code===String(it.bina_item_code||it.item_name||""))||null;
    return {
      description:it.description||"",
      quantity:it.quantity,
      unitPrice:it.unit_price,
      discount:it.discount_pct,
      binaItemCode:it.bina_item_code||it.item_name||"",
      deptName:d?d.name:"",
      lineTotal:it.line_total!=null?it.line_total:null,
    };
  });

  const convert=async(q)=>{
    setBusyId(q.id);
    try{
      const{data:qItems}=await supabaseClient.from("quote_items").select("*").eq("quote_id",q.id).order("line_number");
      const list=Array.isArray(clients)?clients:[];
      const client=list.find(c=>String(c.bina_customer_id)===String(q.bina_cust_id))||null;
      onConvertToOrder?.({
        mode:"order",
        sourceQuoteId:q.id,
        title:q.title||"",
        agent:q.sales_agent||"",
        contactName:q.contact_person||"",
        client,
        bina_cust_id:q.bina_cust_id,
        items:mapQuoteItems(qItems),
      });
    }catch(e){window.alert(e.message||String(e));}
    finally{setBusyId(null);}
  };

  const printQuote=async(q)=>{
    const{data:qItems}=await supabaseClient.from("quote_items").select("*").eq("quote_id",q.id).order("line_number");
    const list=Array.isArray(clients)?clients:[];
    const client=list.find(c=>String(c.bina_customer_id)===String(q.bina_cust_id))||{name:q.bina_cust_name};
    const items=mapQuoteItems(qItems);
    window.openDocumentPDF&&window.openDocumentPDF(hb3BuildPdfDoc({
      kind:"quote",binaDocId:q.bina_doc_id,title:q.title,client,contactName:q.contact_person,items,agent:q.sales_agent,remark:"",paymentTerms:hb3ClientTerms(client),dueDate:"",priceMode:"unit"
    }));
  };

  const remove=async(q)=>{
    if(!window.confirm("למחוק את ההצעה?"))return;
    setBusyId(q.id);
    try{
      const{error}=await supabaseClient.from("quotes").delete().eq("id",q.id);
      if(error)throw error;
      await load();
    }catch(e){window.alert(e.message||String(e));}
    finally{setBusyId(null);}
  };

  const duplicateOrder=o=>{
    const list=Array.isArray(clients)?clients:[];
    const client=list.find(c=>String(c.bina_customer_id)===String(o.bina_cust_id))||null;
    const items=(o.order_items||[]).sort((a,b)=>(a.line_number||0)-(b.line_number||0)).map(it=>{
      const d=HB3_DEPT_FALLBACK.find(x=>x.bina_item_code===String(it.bina_item_code||""))||null;
      return {
        description:it.description||"",
        quantity:it.quantity,
        unitPrice:it.unit_price,
        discount:it.discount_pct,
        binaItemCode:it.bina_item_code||"",
        deptName:d?d.name:"",
        lineTotal:it.line_total,
      };
    });
    onDuplicateOrder?.({
      mode:"order",
      skipFileRequired:true,
      duplicatedFromOrderId:o.bina_doc_id,
      sourceCreatedAt:o.created_at,
      title:o.title||"",
      agent:o.sales_agent||"",
      contactName:o.contact_person||"",
      client,
      bina_cust_id:o.bina_cust_id,
      remark:o.remark||"",
      items,
    });
  };

  const printOrder=o=>{
    const list=Array.isArray(clients)?clients:[];
    const client=list.find(c=>String(c.bina_customer_id)===String(o.bina_cust_id))||{name:o.bina_cust_name};
    const items=(o.order_items||[]).map(it=>({
      description:it.description,quantity:it.quantity,unitPrice:it.unit_price,discount:it.discount_pct,lineTotal:it.line_total
    }));
    window.openDocumentPDF&&window.openDocumentPDF(hb3BuildPdfDoc({
      kind:"order",binaDocId:o.bina_doc_id,title:o.title,client,contactName:o.contact_person,items,agent:o.sales_agent,remark:o.remark||"",paymentTerms:hb3ClientTerms(client),dueDate:"",priceMode:"unit"
    }));
  };

  const sortedOpen=[...openQuotes].sort((a,b)=>ageDays(b)-ageDays(a));
  const hq=histQ.trim().toLowerCase();
  const filteredOrders=!hq?orders:orders.filter(o=>{
    const blob=[o.bina_cust_name,o.title,String(o.bina_doc_id||""),(o.order_items||[]).map(it=>it.description).join(" ")].join(" ").toLowerCase();
    return blob.includes(hq);
  });

  return<div style={{direction:"rtl",maxWidth:960}}>
    <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",gap:12,flexWrap:"wrap",marginBottom:16}}>
      <h1 style={{fontSize:22,fontWeight:900,margin:0,color:"#0E3651"}}>📄 הצעות מחיר</h1>
      <button type="button" onClick={load} style={{padding:"8px 12px",borderRadius:8,border:"1px solid var(--border)",background:"#fff",cursor:"pointer",fontFamily:"inherit"}}>רענן</button>
    </div>
    <div style={{display:"grid",gridTemplateColumns:"repeat(4,1fr)",gap:12,marginBottom:20}}>
      {[{l:"פתוחות",v:openQuotes.length,c:"#2563EB"},{l:"אושרו החודש",v:approvedCount,c:"#16a34a"},{l:"נדחו",v:rejectedCount,c:"#dc2626"},{l:"אחוז המרה",v:convRate+"%",c:"#7c3aed"}].map(k=>
        <div key={k.l} style={{background:"#fff",border:"1px solid var(--border)",borderRadius:12,padding:14,textAlign:"center"}}>
          <div style={{fontSize:28,fontWeight:900,color:k.c}}>{k.v}</div>
          <div style={{fontSize:13,color:"#64748b",fontWeight:600}}>{k.l}</div>
        </div>
      )}
    </div>
    {loading?<div style={{padding:40,textAlign:"center"}}><Spinner sz={28}/> טוען...</div>:
    <div style={{display:"grid",gap:10}}>
      {sortedOpen.map(q=>{
        const d=ageDays(q);
        return<div key={q.id} style={{background:"#fff",border:"1px solid var(--border)",borderRight:"4px solid "+ageColor(d),borderRadius:12,padding:14}}>
          <div style={{display:"flex",justifyContent:"space-between",gap:10,flexWrap:"wrap"}}>
            <div>
              <div style={{fontWeight:800,fontSize:16}}>{q.bina_cust_name||"—"} · {q.title||"הצעה"}</div>
              <div style={{fontSize:13,color:"#64748b",marginTop:4}}>#{q.bina_doc_id||"—"} · {d} ימים · {q.sales_agent||""}</div>
              {q.follow_up_sent_at&&<div style={{fontSize:12,color:"#0d9488",marginTop:4}}>תזכורת נשלחה {new Date(q.follow_up_sent_at).toLocaleDateString("he-IL")}</div>}
            </div>
            <div style={{display:"flex",gap:6,flexWrap:"wrap"}}>
              <button type="button" disabled={busyId===q.id} onClick={()=>convert(q)} style={{padding:"8px 12px",borderRadius:8,border:"none",background:"#16a34a",color:"#fff",fontWeight:700,cursor:"pointer",fontFamily:"inherit",minHeight:40}}>✅ אושרה</button>
              <button type="button" disabled={busyId===q.id} onClick={()=>setStatus(q,"rejected")} style={{padding:"8px 12px",borderRadius:8,border:"1px solid #fecaca",background:"#fff",color:"#dc2626",fontWeight:700,cursor:"pointer",fontFamily:"inherit",minHeight:40}}>❌ נדחתה</button>
              <button type="button" disabled={busyId===q.id} onClick={()=>sendFollowUp(q)} style={{padding:"8px 12px",borderRadius:8,border:"1px solid var(--border)",background:"#fff",cursor:"pointer",fontFamily:"inherit",minHeight:40}}>📧 תזכורת</button>
              <button type="button" onClick={()=>printQuote(q)} style={{padding:"8px 12px",borderRadius:8,border:"1px solid var(--border)",background:"#fff",cursor:"pointer",fontFamily:"inherit",minHeight:40}}>📄 PDF</button>
              <button type="button" onClick={()=>setViewQ(q)} style={{padding:"8px 12px",borderRadius:8,border:"1px solid var(--border)",background:"#fff",cursor:"pointer",fontFamily:"inherit",minHeight:40}}>👁️</button>
              <button type="button" disabled={busyId===q.id} onClick={()=>remove(q)} style={{padding:"8px 12px",borderRadius:8,border:"none",background:"#fee2e2",color:"#dc2626",cursor:"pointer",fontFamily:"inherit",minHeight:40}}>🗑️</button>
            </div>
          </div>
        </div>;
      })}
      {!sortedOpen.length&&<div style={{padding:40,textAlign:"center",color:"#94a3b8"}}>אין הצעות פתוחות</div>}
    </div>}
    {viewQ&&<Modal open={true} onClose={()=>setViewQ(null)} title={"הצעה — "+(viewQ.title||"")} maxWidth={520}>
      <div style={{fontSize:14,lineHeight:1.7}}>
        <div><strong>לקוח:</strong> {viewQ.bina_cust_name}</div>
        <div><strong>בינה:</strong> #{viewQ.bina_doc_id||"—"}</div>
        <div><strong>סטטוס:</strong> {viewQ.quote_status||"—"}</div>
        <div><strong>נשלחה:</strong> {viewQ.sent_at?new Date(viewQ.sent_at).toLocaleString("he-IL"):"—"}</div>
        <div><strong>סוכן:</strong> {viewQ.sales_agent||"—"}</div>
        <div><strong>איש קשר:</strong> {viewQ.contact_person||"—"}</div>
      </div>
    </Modal>}

    <h2 style={{fontSize:18,fontWeight:800,margin:"28px 0 12px",color:"#0E3651"}}>היסטוריית הזמנות</h2>
    <input className="hb3-inp" style={{maxWidth:420,marginBottom:12}} placeholder="חיפוש: לקוח / מספר הזמנה / תיאור פריט" value={histQ} onChange={e=>setHistQ(e.target.value)}/>
    <div style={{display:"grid",gap:8}}>
      {filteredOrders.map(o=><div key={o.id} style={{background:"#fff",border:"1px solid var(--border)",borderRadius:12,padding:12,display:"flex",justifyContent:"space-between",gap:10,flexWrap:"wrap"}}>
        <div>
          <div style={{fontWeight:800}}>#{o.bina_doc_id||"—"} · {o.bina_cust_name||"—"}</div>
          <div style={{fontSize:13,color:"#64748b",marginTop:4}}>{o.title||""} · {(o.order_items||[]).length} פריטים · {o.created_at?new Date(o.created_at).toLocaleDateString("he-IL"):""}</div>
        </div>
        <div style={{display:"flex",gap:6}}>
          <button type="button" onClick={()=>duplicateOrder(o)} style={{padding:"8px 12px",borderRadius:8,border:"1px solid #BFE6E3",background:"#EDF8F7",color:"#0E3651",fontWeight:700,cursor:"pointer",fontFamily:"inherit",minHeight:40}}>📋 שכפל</button>
          <button type="button" onClick={()=>printOrder(o)} style={{padding:"8px 12px",borderRadius:8,border:"1px solid var(--border)",background:"#fff",cursor:"pointer",fontFamily:"inherit",minHeight:40}}>📄 PDF</button>
        </div>
      </div>)}
      {!filteredOrders.length&&<div style={{padding:24,textAlign:"center",color:"#94a3b8"}}>אין הזמנות תואמות</div>}
    </div>
  </div>;
}
